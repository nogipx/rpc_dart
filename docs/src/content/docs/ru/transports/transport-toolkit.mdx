## Архитектурные принципы

1. **HTTP/2 совместимость** - Stream ID management согласно RFC 7540
2. **Zero-copy оптимизации** - поддержка прямой передачи объектов (InMemory transport)
3. **Модульность** - каждый компонент может использоваться независимо
4. **Расширяемость** - миксины и интерфейсы для добавления функциональности
5. **Отказоустойчивость** - автоматическое переподключение и мониторинг здоровья

Весь инструментарий находится в основном пакете `rpc_dart` и готов к использованию без дополнительных зависимостей.

### RpcBaseTransport

    ```dart
      // Соединение с внешней системой
      late WebSocket _webSocket;
      
      MyCustomTransport({required bool isClient}) : super(isClient: isClient);

      @override
      bool get supportsZeroCopy => false;

      @override
      Future<void> doConnect() async {
        // Ваша логика подключения
        _webSocket = await WebSocket.connect('ws://localhost:8080/rpc');
        
        // Настройка прослушивания входящих сообщений
        _webSocket.listen((data) {
          final message = _deserializeWebSocketMessage(data);
          incomingController.add(message);
        });
        
        print('WebSocket транспорт подключен');
      }

      @override
      Future<void> doSendMetadata(int streamId, RpcMetadata metadata, {bool endStream = false}) async {
        // Отправка метаданных через WebSocket
        final frame = _createMetadataFrame(streamId, metadata, endStream);
        _webSocket.add(frame);
      }

      @override
      Future<void> doSendMessage(int streamId, Uint8List data, {bool endStream = false}) async {
        // Отправка данных через WebSocket
        final frame = _createDataFrame(streamId, data, endStream);
        _webSocket.add(frame);
      }

      @override
      Future<void> doFinishSending(int streamId) async {
        // Завершение отправки для stream
        final frame = _createEndFrame(streamId);
        _webSocket.add(frame);
      }

      @override
      Future<void> doClose() async {
        // Закрытие соединения
        await _webSocket.close();
        print('WebSocket транспорт закрыт');
      }

      // Вспомогательные методы для работы с протоколом
      Uint8List _createMetadataFrame(int streamId, RpcMetadata metadata, bool endStream) {
        // Реализация создания фрейма метаданных

        // Реализация создания фрейма данных
        return Uint8List.fromList([/* фрейм данных */]);
      }

      RpcTransportMessage _deserializeWebSocketMessage(dynamic data) {
        // Десериализация входящих WebSocket сообщений
        // ...
        return RpcTransportMessage(streamId: 1, payload: Uint8List(0));
      }
    }
    ```

## Миксины функциональности

### RpcZeroCopySupport

Добавляет поддержку zero-copy операций для транспортов в пределах одного процесса:

<Tabs>
  <TabItem label="Реализация">
    ```dart
    class MyInMemoryTransport extends RpcBaseTransport with RpcZeroCopySupport {
      final Map<String, MyInMemoryTransport> _peers = {};
      
      MyInMemoryTransport({required bool isClient, required String peerId}) 
        : super(isClient: isClient) {
        _peers[peerId] = this;
      }

      @override
      bool get supportsZeroCopy => true;

      @override
      Future<void> doSendDirectObject(int streamId, Object object, {bool endStream = false}) async {
        // Прямая передача объекта без сериализации
        final targetPeer = _peers['target']; // Получение целевого транспорта
        
        targetPeer?.incomingController.add(RpcTransportMessage.withDirectObject(
          streamId: streamId,
          directPayload: object,
          isEndOfStream: endStream,
        ));
        
        print('Объект передан напрямую: ${object.runtimeType}');
      }

      // Остальные методы реализации...
      @override
      Future<void> doConnect() async {
        // In-memory соединение всегда "подключено"
      }

      @override
      Future<void> doSendMessage(int streamId, Uint8List data, {bool endStream = false}) async {
        // Можно также поддерживать обычную сериализацию
        await doSendDirectObject(streamId, data, endStream: endStream);
      }
    }
    ```
  </TabItem>
  
  <TabItem label="Использование">
    ```dart
    void main() async {
      final clientTransport = MyInMemoryTransport(isClient: true, peerId: 'client');
      final serverTransport = MyInMemoryTransport(isClient: false, peerId: 'server');
      
      // Настройка пиринга
      clientTransport._peers['target'] = serverTransport;
      serverTransport._peers['target'] = clientTransport;
      
      await clientTransport.connect();
      await serverTransport.connect();
      
      // Zero-copy передача объекта
      final complexObject = {
        'data': [1, 2, 3, 4, 5],
        'metadata': {'timestamp': DateTime.now(), 'version': '1.0'},
        'nested': {'deep': {'value': 42}},
      };
      
      final streamId = clientTransport.createStream();
      
      // Объект передается по ссылке, без сериализации!
      await clientTransport.sendDirectObject(streamId, complexObject);
      
      // Получение на серверной стороне
      serverTransport.getMessagesForStream(streamId).listen((message) {
        if (message.isDirect) {
          final receivedObject = message.directPayload as Map<String, dynamic>;
          print('Получен объект: $receivedObject');
          // receivedObject ссылается на тот же объект в памяти!
        }
      });
    }
    ```

### RpcMessageBuffering

Добавляет буферизацию и батчинг сообщений для оптимизации производительности:

<Tabs>
  <TabItem label="Конфигурация">
    ```dart
    class MyBufferedTransport extends RpcBaseTransport with RpcMessageBuffering {
      late TcpSocket _socket;
      
      MyBufferedTransport({required bool isClient}) : super(isClient: isClient) {
        // Настройка буферизации
        configureBuffering(
          maxBufferSize: 1024 * 1024,           // 1MB буфер
          flushInterval: Duration(milliseconds: 100), // Сброс каждые 100мс
          enableBatching: true,                  // Включить батчинг
        );
      }

      @override
      Future<void> doSendBufferedMessages(List<BufferedMessage> messages) async {
        if (_socket.isClosed) throw StateError('Сокет закрыт');
        
        // Отправка всех сообщений в одном TCP пакете
        final batch = BytesBuilder();
        
        for (final message in messages) {
          // Создание фрейма для каждого сообщения
          final frame = _createFrame(message.streamId, message.data, message.endStream);
          batch.add(frame);
        }
        
        await _socket.write(batch.toBytes());
        
        print('Отправлен батч из ${messages.length} сообщений '
              '(${batch.length} байт)');
      }

      @override
      Future<void> doConnect() async {
        _socket = await TcpSocket.connect('localhost', 8080);
      }

      @override
      Future<void> doSendMessage(int streamId, Uint8List data, {bool endStream = false}) async {
        // Сообщения автоматически буферизуются
        await bufferMessage(streamId, data, endStream);
      }

      Uint8List _createFrame(int streamId, Uint8List data, bool endStream) {
        // Создание фрейма протокола
        final frame = BytesBuilder();
        frame.add(_encodeStreamId(streamId));
        frame.add(_encodeFlags(endStream));
        frame.add(_encodeLength(data.length));
        frame.add(data);
        return frame.toBytes();
      }
    }
    ```

### RpcAutoReconnect

Добавляет автоматическое переподключение с экспоненциальной задержкой:

<Tabs>
  <TabItem label="Настройка">
    ```dart
    class MyReliableTransport extends RpcBaseTransport with RpcAutoReconnect {
      String _serverUrl;
      HttpClient? _httpClient;
      
      MyReliableTransport({required bool isClient, required String serverUrl}) 
        : _serverUrl = serverUrl, super(isClient: isClient) {
        
        // Настройка переподключения
        configureReconnection(
          maxAttempts: 10,                       // Максимум 10 попыток
          initialDelay: Duration(seconds: 1),    // Начальная задержка 1с
          maxDelay: Duration(seconds: 60),       // Максимальная задержка 60с
          backoffMultiplier: 2.0,                // Удвоение задержки
        );
      }

      @override
      Future<bool> checkConnection() async {
        if (_httpClient == null) return false;
        
        try {
          // Простая проверка ping
          final request = await _httpClient!.get(Uri.parse('$_serverUrl/ping'));
          final response = await request.close();
          return response.statusCode == 200;
        } catch (e) {
          return false;
        }
      }

      @override
      Future<void> attemptReconnection() async {
        print('Попытка переподключения к $_serverUrl...');
        
        _httpClient?.close(force: true);
        _httpClient = null;
        
        await doConnect();
        
        print('Переподключение успешно!');
      }

      @override
      Future<void> doConnect() async {
        _httpClient = HttpClient();
        
        // Проверка доступности сервера
        final request = await _httpClient!.get(Uri.parse('$_serverUrl/connect'));
        final response = await request.close();
        
        if (response.statusCode != 200) {
          throw Exception('Не удалось подключиться: ${response.statusCode}');
        }
        
        // Настройка прослушивания разрывов соединения
        _httpClient!.connectionTimeout = Duration(seconds: 30);
        
        print('HTTP транспорт подключен к $_serverUrl');
      }

      @override
      Future<void> doSendMessage(int streamId, Uint8List data, {bool endStream = false}) async {
        if (_httpClient == null) {
          throw StateError('Клиент не подключен');
        }
        
        try {
          final request = await _httpClient!.post(Uri.parse('$_serverUrl/rpc'));
          request.headers.set('stream-id', streamId.toString());
          request.headers.set('end-stream', endStream.toString());
          request.add(data);
          
          final response = await request.close();
          
          if (response.statusCode != 200) {
            throw Exception('Ошибка отправки: ${response.statusCode}');
          }
          
        } catch (e) {
          // Автоматическое переподключение при ошибке
          handleConnectionError(e);
          rethrow;
        }
      }
    }
    ```
  </TabItem>
  
  <TabItem label="Обработка ошибок">
    ```dart

    class ConfigValidator {
      static void validateTransportConfig(Map<String, dynamic> config) {
        // Проверка обязательных полей
        if (!config.containsKey('isClient')) {
          throw ArgumentError('Не указана роль транспорта (isClient)');
        }
        
        // Валидация буферизации
        if (config.containsKey('buffering')) {
          final buffering = config['buffering'] as Map<String, dynamic>;
          final maxSize = buffering['maxBufferSize'] as int?;
          final flushInterval = buffering['flushInterval'] as Duration?;
          
          if (maxSize != null && maxSize <= 0) {
            throw ArgumentError('Размер буфера должен быть положительным');
          }
          
          if (flushInterval != null && flushInterval.isNegative) {
            throw ArgumentError('Интервал сброса не может быть отрицательным');
          }
        }
        
        // Валидация переподключения
        if (config.containsKey('reconnection')) {
          final reconnection = config['reconnection'] as Map<String, dynamic>;
          final maxAttempts = reconnection['maxAttempts'] as int?;
          final backoffMultiplier = reconnection['backoffMultiplier'] as double?;
          
          if (maxAttempts != null && maxAttempts < 0) {
            throw ArgumentError('Количество попыток не может быть отрицательным');
          }
          
          if (backoffMultiplier != null && backoffMultiplier < 1.0) {
            throw ArgumentError('Множитель задержки должен быть >= 1.0');
          }
        }
        
        // Валидация производительности
        if (config.containsKey('performance')) {
          final performance = config['performance'] as Map<String, dynamic>;
          final maxStreams = performance['maxConcurrentStreams'] as int?;
          final frameSize = performance['messageFrameSize'] as int?;
          
          if (maxStreams != null && maxStreams <= 0) {
            throw ArgumentError('Максимальное количество стримов должно быть положительным');
          }
          
          if (frameSize != null && (frameSize < 1024 || frameSize > 1024 * 1024)) {
            throw ArgumentError('Размер фрейма должен быть между 1KB и 1MB');
          }
        }
      }
    }
    ```

### Полнофункциональный WebSocket транспорт

Пример создания полнофункционального WebSocket транспорта с использованием всех возможностей инструментария:

<Tabs>
  <TabItem label="Реализация транспорта">
    ```dart
    import 'dart:io';
    import 'package:rpc_dart/rpc_dart.dart';

    class RpcWebSocketTransport extends RpcBaseTransport 
        with RpcMessageBuffering, RpcAutoReconnect {
      
      final String _url;
      final Map<String, String> _headers;
      WebSocket? _webSocket;
      
      RpcWebSocketTransport({
        required bool isClient,
        required String url,
        Map<String, String> headers = const {},
        Map<String, dynamic>? config,
      }) : _url = url,
           _headers = headers,
           super(isClient: isClient) {
        
        // Применение конфигурации
        if (config != null) {
          _applyConfig(config);
        }
      }

      @override
      bool get supportsZeroCopy => false;

      void _applyConfig(Map<String, dynamic> config) {
        // Настройка буферизации
        if (config.containsKey('buffering')) {
          final bufferConfig = config['buffering'] as Map<String, dynamic>;
          configureBuffering(
            maxBufferSize: bufferConfig['maxBufferSize'] ?? 64 * 1024,
            flushInterval: bufferConfig['flushInterval'] ?? Duration(milliseconds: 100),
            enableBatching: bufferConfig['enableBatching'] ?? true,
          );
        }
        
        // Настройка переподключения
        if (config.containsKey('reconnection')) {
          final reconnectConfig = config['reconnection'] as Map<String, dynamic>;
          configureReconnection(
            maxAttempts: reconnectConfig['maxAttempts'] ?? 5,
            initialDelay: reconnectConfig['initialDelay'] ?? Duration(seconds: 1),
            maxDelay: reconnectConfig['maxDelay'] ?? Duration(seconds: 30),
            backoffMultiplier: reconnectConfig['backoffMultiplier'] ?? 2.0,
          );
        }
      }

      @override
      Future<void> doConnect() async {
        print('Подключение к WebSocket: $_url');
        
        _webSocket = await WebSocket.connect(_url, headers: _headers);
        
        // Настройка прослушивания сообщений
        _webSocket!.listen(
          _handleWebSocketMessage,
          onError: (error) {
            print('Ошибка WebSocket: $error');
            handleConnectionError(error);
          },
          onDone: () {
            print('WebSocket соединение закрыто');
            handleConnectionError(Exception('Соединение разорвано'));
          },
        );
        
        print('WebSocket транспорт подключен');
      }

      @override
      Future<bool> checkConnection() async {
        return _webSocket != null && 
               _webSocket!.readyState == WebSocket.open;
      }

      @override
      Future<void> attemptReconnection() async {
        print('Переподключение WebSocket...');
        
        _webSocket?.close();
        _webSocket = null;
        
        await doConnect();
      }

      @override
      Future<void> doSendMetadata(int streamId, RpcMetadata metadata, {bool endStream = false}) async {
        final frame = _createMetadataFrame(streamId, metadata, endStream);
        await bufferMessage(streamId, frame, endStream);
      }

      @override
      Future<void> doSendMessage(int streamId, Uint8List data, {bool endStream = false}) async {
        final frame = _createDataFrame(streamId, data, endStream);
        await bufferMessage(streamId, frame, endStream);
      }

      @override
      Future<void> doSendBufferedMessages(List<BufferedMessage> messages) async {
        if (_webSocket == null || _webSocket!.readyState != WebSocket.open) {
          throw StateError('WebSocket не подключен');
        }
        
        // Создание батча сообщений
        final batch = BytesBuilder();
        for (final message in messages) {
          batch.add(message.data);
        }
        
        _webSocket!.add(batch.toBytes());
        
        print('Отправлен батч из ${messages.length} сообщений');
      }

      @override
      Future<void> doFinishSending(int streamId) async {
        final frame = _createEndStreamFrame(streamId);
        await bufferMessage(streamId, frame, true);
        await flushBuffer(); // Принудительный сброс
      }

      @override
      Future<void> doClose() async {
        await flushBuffer();
        await _webSocket?.close();
        _webSocket = null;
        print('WebSocket транспорт закрыт');
      }

      void _handleWebSocketMessage(dynamic data) {
        try {
          final bytes = data as Uint8List;
          final message = _parseWebSocketFrame(bytes);
          
          if (message != null) {
            incomingController.add(message);
          }
          
        } catch (e) {
          print('Ошибка обработки WebSocket сообщения: $e');
        }
      }

      Uint8List _createMetadataFrame(int streamId, RpcMetadata metadata, bool endStream) {
        final serializedMetadata = RpcMessageSerialization.toJson(
          RpcTransportMessage.withMetadata(
            streamId: streamId,
            metadata: metadata,
            isEndOfStream: endStream,
          ),
        );
        
        final frameData = utf8.encode(jsonEncode(serializedMetadata));
        return _wrapInFrame(WebSocketFrameType.metadata, frameData);
      }

      Uint8List _createDataFrame(int streamId, Uint8List data, bool endStream) {
        final message = RpcTransportMessage.withPayload(
          streamId: streamId,
          payload: data,
          isEndOfStream: endStream,
        );
        
        final frameData = RpcMessageSerialization.toBinary(message);
        return _wrapInFrame(WebSocketFrameType.data, frameData);
      }

      Uint8List _createEndStreamFrame(int streamId) {
        final message = RpcTransportMessage(
          streamId: streamId,
          isEndOfStream: true,
        );
        
        final frameData = RpcMessageSerialization.toBinary(message);
        return _wrapInFrame(WebSocketFrameType.endStream, frameData);
      }

      Uint8List _wrapInFrame(WebSocketFrameType type, Uint8List data) {
        final frame = BytesBuilder();
        frame.addByte(type.index);                    // Тип фрейма (1 байт)
        frame.add(RpcTransportUtils.encodeLength(data.length)); // Длина (4 байта)
        frame.add(data);                              // Данные
        return frame.toBytes();
      }

      RpcTransportMessage? _parseWebSocketFrame(Uint8List data) {
        if (data.length < 5) return null; // Минимальный размер фрейма
        
        final type = WebSocketFrameType.values[data[0]];
        final length = RpcTransportUtils.decodeLength(data.sublist(1, 5));
        
        if (data.length < 5 + length) return null; // Неполный фрейм
        
        final frameData = data.sublist(5, 5 + length);
        
        switch (type) {
          case WebSocketFrameType.metadata:
            final jsonData = jsonDecode(utf8.decode(frameData));
            return RpcMessageSerialization.fromJson(jsonData);
            
          case WebSocketFrameType.data:
          case WebSocketFrameType.endStream:
            return RpcMessageSerialization.fromBinary(frameData);
        }
      }
    }

    enum WebSocketFrameType {
      metadata,
      data,
      endStream,
    }
    ```
